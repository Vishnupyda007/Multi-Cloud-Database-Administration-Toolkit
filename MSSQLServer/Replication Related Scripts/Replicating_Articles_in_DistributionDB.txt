USE [Distribution]
SELECT 
    cmds.article_id,
    Ma.article,
    CAST(cmds.partial_command AS TINYINT),
    cmds.command,
    cmds.xact_seqno,
    cmds.xact_seqno,
    0,
    cmds.command_id,
    cmds.type,
    orgs.srvname,
    orgs.dbname,
    CAST(cmds.hashkey AS SMALLINT),
    orgs.publication_id,
    orgs.dbversion,
    cmds.originator_lsn
FROM 
    MSrepl_commands cmds 
LEFT JOIN 
    MSrepl_originators orgs 
ON 
    cmds.originator_id = orgs.id
LEFT JOIN
    MSarticles Ma
ON
    cmds.article_id = Ma.article_id
WHERE 
    cmds.xact_seqno >= 0x00005CC7000115DA000E 
    AND cmds.xact_seqno <= 0x00005CC7000115DA000E 
ORDER BY 
    cmds.originator_id, 
    cmds.publisher_database_id, 
    cmds.xact_seqno, 
    cmds.article_id, 
    cmds.command_id ASC
