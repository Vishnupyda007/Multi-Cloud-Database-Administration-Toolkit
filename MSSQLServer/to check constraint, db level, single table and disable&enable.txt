/*To check constraint status*/

SELECT	OBJECT_NAME(RKEYID) AS REFERENCED_TABLE_NAME,
		OBJECT_NAME(FKEYID) AS CHILD_TABLE_NAME,
OBJECT_NAME(CONSTID) AS CONSTRAINT_NAME,
COL_NAME(FKEYID, FKEY) AS COLUMN_NAME,
COL_NAME(RKEYID, RKEY) AS REFERENCED_COLUMN_NAME,
		(CASE
WHEN OBJECTPROPERTY(CONSTID, 'CNSTISDISABLED') = 0 THEN 'ENABLED'
ELSE 'DISABLED'
END) AS STATUS
FROM SYSFOREIGNKEYS
ORDER BY CHILD_TABLE_NAME, CONSTRAINT_NAME,REFERENCED_TABLE_NAME, KEYNO;
