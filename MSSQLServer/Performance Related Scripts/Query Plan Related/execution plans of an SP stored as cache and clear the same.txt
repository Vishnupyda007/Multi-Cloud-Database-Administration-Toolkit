------------Query to get the execution plans of an SP stored as cache------------
 
SELECT

    cp.plan_handle,

    sp.name AS StoredProcedureName

FROM

    sys.dm_exec_cached_plans AS cp

    CROSS APPLY sys.dm_exec_sql_text(cp.plan_handle) AS st

    JOIN sys.procedures AS sp ON st.objectid = sp.object_id

WHERE
    sp.name = 'procedure_Name';

 
----------------to clean cache of execution plan of an SP-------------------

DBCC FREEPROCCACHE (0x02000000A89D650F91F4BF4DAD1C4B6F);
