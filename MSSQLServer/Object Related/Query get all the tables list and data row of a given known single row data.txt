DECLARE @ColumnName NVARCHAR(100) = 'AssociateID';
DECLARE @SearchValue NVARCHAR(100) = 'A12345';
DECLARE @SQL NVARCHAR(MAX) = '';

SELECT @SQL = @SQL + '
IF EXISTS (SELECT 1 FROM [' + s.name + '].[' + t.name + '] WHERE [' + c.name + '] = ''' + @SearchValue + ''')
BEGIN
    PRINT ''Found in table: [' + s.name + '].[' + t.name + ']'';
    SELECT ''' + s.name + '.' + t.name + ''' AS TableName, * 
    FROM [' + s.name + '].[' + t.name + '] 
    WHERE [' + c.name + '] = ''' + @SearchValue + ''';
END;
'
FROM sys.columns c
JOIN sys.tables t ON c.object_id = t.object_id
JOIN sys.schemas s ON t.schema_id = s.schema_id
WHERE c.name = @ColumnName;

EXEC sp_executesql @SQL;
