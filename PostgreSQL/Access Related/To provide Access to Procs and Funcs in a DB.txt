this is a very common Cloud SQL PostgreSQL gotcha, especially once you start using procedures (PostgreSQL 11+).
Below is the clean, production-safe way to give default access to procedures automatically so you don’t have to GRANT every time 🚀
🎯 Goal
Ensure that any new PROCEDURE created in a schema automatically gets EXECUTE permission for a role (app/user).
✅ Recommended Script (Default Privileges)
Run this once, as the procedure owner role (very important).
🔹 Grant default EXECUTE on future procedures
ALTER DEFAULT PRIVILEGES
IN SCHEMA public
GRANT EXECUTE ON ROUTINES TO app_user;
📌 Notes:
ROUTINES covers both FUNCTIONS and PROCEDURES
Applies only to objects created AFTER this command
Replace:
public → your schema name
app_user → your application role
🔐 If procedures are created by a specific role
Default privileges are role-specific.
If procedures are created by proc_owner:
ALTER DEFAULT PRIVILEGES
FOR ROLE proc_owner
IN SCHEMA public
GRANT EXECUTE ON ROUTINES TO app_user;
✅ This is the most correct pattern in Cloud SQL.
🧠 Existing Procedures (one-time fix)
Default privileges do NOT apply retroactively.
Grant access to existing procedures:
GRANT EXECUTE ON ALL ROUTINES IN SCHEMA public TO app_user;
🏗️ Multiple Schemas (recommended for prod)
ALTER DEFAULT PRIVILEGES
FOR ROLE proc_owner
IN SCHEMA reporting
GRANT EXECUTE ON ROUTINES TO app_user;
Repeat per schema (PostgreSQL has no global default privilege switch).
🔍 Verify Permissions
SELECT routine_schema,
       routine_name,
       privilege_type
FROM information_schema.role_routine_grants
WHERE grantee = 'app_user';

WHERE n.nspname = 'public'
AND NOT EXISTS (
  SELECT 1
  FROM information_schema.role_routine_grants g
  WHERE g.routine_name = p.proname

ALTER DEFAULT PRIVILEGES
FOR ROLE "prodgke-devops-serviceacc@cb0228600a-citprod-gc.iam",postgres
IN SCHEMA public
GRANT EXECUTE ON ROUTINEs  TO "clientdrift-5043-prod-sa@cb0228600a-citprod-gc.iam";
--revoke execute on all routines in schema public from public;


DO $$
DECLARE
    -- Variable to hold the record for each routine found
    routine_record RECORD;
    -- The role you are granting privileges TO
    target_role TEXT := 'clientdrift-5043-prod-sa@cb0228600a-citprod-gc.iam';
BEGIN
    -- Loop through all routines (functions and procedures) in the 'public' schema
    -- that are owned by the current user.
    FOR routine_record IN
        SELECT
            p.proname AS routine_name,
            pg_get_function_identity_arguments(p.oid) AS routine_args,
            p.prokind -- 'f' for function, 'p' for procedure
        FROM
            pg_proc p
        JOIN
            pg_namespace n ON p.pronamespace = n.oid
        JOIN
            pg_roles r ON p.proowner = r.oid
        WHERE
            n.nspname = 'public'  -- Only in the public schema
            AND r.rolname in ('prodgke-devops-serviceacc@cb0228600a-citprod-gc.iam','postgres') -- IMPORTANT: Only routines owned by the user running this script
            AND p.prokind IN ('f', 'p') -- Only select functions and procedures
    LOOP
        -- Check if it's a FUNCTION
        IF routine_record.prokind = 'f' THEN
            RAISE NOTICE 'Granting EXECUTE on FUNCTION public.%(%) to %', routine_record.routine_name, routine_record.routine_args, target_role;
            EXECUTE format(
                'GRANT EXECUTE ON FUNCTION public.%I(%s) TO %I',
                routine_record.routine_name,
                routine_record.routine_args,
                target_role
            );
        -- Check if it's a PROCEDURE
        ELSIF routine_record.prokind = 'p' THEN
            RAISE NOTICE 'Granting EXECUTE on PROCEDURE public.%(%) to %', routine_record.routine_name, routine_record.routine_args, target_role;
            EXECUTE format(
                'GRANT EXECUTE ON PROCEDURE public.%I(%s) TO %I',
                routine_record.routine_name,
                routine_record.routine_args,
                target_role
            );
        END IF;
    END LOOP;
END $$;

DO $$
DECLARE
    -- Variable to hold the record for each routine found
    routine_record RECORD;
    -- The role you are granting privileges TO
    target_role TEXT := 'va-chatanalysis-5022-np-sa@cb0104074a-citnonprod-gc.iam';
BEGIN
    -- Loop through all routines (functions and procedures) owned by the specified roles
    FOR routine_record IN
        SELECT
            p.proname AS routine_name,
            n.nspname AS schema_name,
            p.prokind,
            pg_get_function_identity_arguments(p.oid) AS routine_identity_args,
            
            -- This now works because the set-returning function (unnest) is in the FROM clause,
            -- and the aggregate (string_agg) is in the SELECT list.
            string_agg(format_type(t.arg_oid, NULL), ', ' ORDER BY t.arg_num) AS routine_simple_signature

        FROM
            pg_proc p
        JOIN
            pg_namespace n ON p.pronamespace = n.oid
        JOIN
            pg_roles r ON p.proowner = r.oid
        -- Use a LEFT JOIN LATERAL to handle routines with zero arguments correctly.
        -- This "expands" the array of argument types into separate rows.
        LEFT JOIN LATERAL
            unnest(p.proargtypes) WITH ORDINALITY AS t(arg_oid, arg_num) ON true
        WHERE
            n.nspname != 'DBAdmin'  -- Exclude the DBAdmin schema
            AND r.rolname IN ('npgke-devops-serviceacc@cb0104074a-citnonprod-gc.iam', 'OneC_5022') -- Filter by owner roles
            AND p.prokind IN ('f', 'p') -- Only select functions and procedures
        GROUP BY
            -- We must group by all non-aggregated columns to collapse the rows back.
            p.oid, n.nspname, p.proname, p.prokind
    LOOP
        -- The logic inside the loop remains the same
        IF routine_record.prokind = 'f' THEN
            RAISE NOTICE 'Granting EXECUTE on FUNCTION %.%(%) to %', routine_record.schema_name, routine_record.routine_name, routine_record.routine_identity_args, target_role;
            EXECUTE format(
                'GRANT EXECUTE ON FUNCTION %I.%I(%s) TO %I',
                routine_record.schema_name,
                routine_record.routine_name,
                COALESCE(routine_record.routine_simple_signature, ''), -- Use COALESCE for functions with no arguments
                target_role
            );
        ELSIF routine_record.prokind = 'p' THEN
            RAISE NOTICE 'Granting EXECUTE on PROCEDURE %.%(%) to %', routine_record.schema_name, routine_record.routine_name, routine_record.routine_identity_args, target_role;
            EXECUTE format(
                'GRANT EXECUTE ON PROCEDURE %I.%I(%s) TO %I',
                routine_record.schema_name,
                routine_record.routine_name,
                COALESCE(routine_record.routine_simple_signature, ''), -- Use COALESCE for procedures with no arguments
                target_role
            );
        END IF;
    END LOOP;
END $$;
